1. Check the core Lab 12 resources:
   kubectl get svc,configmap,pvc -n {{ .Release.Namespace }}

2. Verify the mounted config file:
   kubectl exec -n {{ .Release.Namespace }} deploy/{{ include "common.fullname" . }} \
     -c {{ include "common.name" . }} \
     -- cat {{ include "devops-info-python.configFilePath" . }}

3. Verify the injected environment variables:
   kubectl exec -n {{ .Release.Namespace }} deploy/{{ include "common.fullname" . }} \
     -c {{ include "common.name" . }} \
     -- sh -lc "env | grep -E '^(APP_NAME|APP_ENV|APP_MESSAGE|HOST|PORT|LOG_LEVEL|APP_CONFIG_PATH|VISITS_FILE_PATH|FEATURE_.*)=' | sort"

4. Verify the application endpoints:
   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "common.fullname" . }} 8080:{{ .Values.service.port }}
   curl http://127.0.0.1:8080/
   curl http://127.0.0.1:8080/health
   curl http://127.0.0.1:8080/visits

5. Verify persistent storage survives a pod restart:
   kubectl exec -n {{ .Release.Namespace }} deploy/{{ include "common.fullname" . }} \
     -c {{ include "common.name" . }} \
     -- cat {{ include "devops-info-python.visitsFilePath" . }}
   kubectl delete pod -n {{ .Release.Namespace }} $(kubectl get pod -n {{ .Release.Namespace }} -l app.kubernetes.io/instance={{ .Release.Name }} -o jsonpath='{.items[0].metadata.name}')
   kubectl rollout status deployment/{{ include "common.fullname" . }} -n {{ .Release.Namespace }} --timeout=180s
   kubectl exec -n {{ .Release.Namespace }} deploy/{{ include "common.fullname" . }} \
     -c {{ include "common.name" . }} \
     -- cat {{ include "devops-info-python.visitsFilePath" . }}

6. Trigger a rollout when ConfigMap values change through Helm:
   helm upgrade {{ .Release.Name }} <same-chart-reference> -n {{ .Release.Namespace }} --reuse-values --set config.logLevel=INFO
   kubectl rollout status deployment/{{ include "common.fullname" . }} -n {{ .Release.Namespace }}
