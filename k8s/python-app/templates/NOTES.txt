1. Get the release status:
   helm status {{ .Release.Name }}

2. Check the Kubernetes objects:
   kubectl get po,sts,svc,pvc -l app.kubernetes.io/instance={{ .Release.Name }}

3. If the service type is NodePort, get the URL with Minikube:
   minikube service {{ include "python-app.fullname" . }} --url

4. If StatefulSet mode is enabled, verify pod DNS names:
   kubectl get pod {{ include "python-app.fullname" . }}-0
   kubectl exec -it {{ include "python-app.fullname" . }}-0 -- nslookup {{ include "python-app.fullname" . }}-1.{{ include "python-app.headlessServiceName" . }}
